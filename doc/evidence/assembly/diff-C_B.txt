+import { DomainError } from '@svgframe/core';
+import { BoxShape } from './BoxShape.js';
+import type {
+  AttributeEditContext,
+  AttributeHandler,
+  AttributeRequest,
+  AttributeResult,
+  ReadonlyShape,
+} from '@svgframe/core';
+
+/**
+ * Domain attribute semantics for boxes: `fill` is propagated to the necessary
+ * rectangle and the plugin-only convenience attribute `label` rewrites the
+ * necessary text. Both run on the candidate document, so one `set` remains a single
+ * undo unit even though a part is modified. `handles` is a pure query and only
+ * claims boxes: a `label` on any other shape keeps falling through to the Phase 1
+ * attribute whitelist and therefore reports `unknown attribute: label`.
+ */
+export class BoxAttributeHandler implements AttributeHandler {
+  readonly name = 'box-attributes';
+
+  handles(target: ReadonlyShape, attribute: string): boolean {
+    return target instanceof BoxShape && (attribute === 'fill' || attribute === 'label');
+  }
+
+  apply(context: AttributeEditContext, request: AttributeRequest): AttributeResult {
+    const box = context.document.require(request.targetId);
+    if (!(box instanceof BoxShape)) throw new DomainError(`not a box: ${request.targetId}`);
+    if (request.attribute === 'fill') {
+      const rect = box.rect();
+      rect.setAttribute('fill', request.value);
+      return { message: `box#${box.id} fill propagated to ${rect.id}` };
+    }
+    if (request.attribute === 'label') {
+      const label = box.label();
+      label.setAttribute('text', request.value);
+      return { message: `box#${box.id} label updated: ${JSON.stringify(request.value)}` };
+    }
+    throw new DomainError(`unknown attribute: ${request.attribute}`);
+  }
+}

+import { DomainError, Numbers } from '@svgframe/core';
+import { BoxShape } from './BoxShape.js';
+import type {
+  Shape,
+  ShapeCreationRequest,
+  ShapeCreationService,
+  ShapeProvider,
+  ShapeRestoreService,
+  SvgElementView,
+} from '@svgframe/core';
+
+/** Role template fixed by Phase 1 §6.1B: rect at (x,y,w,h), label at (x+20, y+h/2+5). */
+const DEFAULT_FILL = '#4a90d9';
+const LABEL_OFFSET_X = 20;
+const LABEL_BASELINE_OFFSET = 5;
+
+/**
+ * Provider for the `box` composite type.
+ *
+ * `create` owns the Phase 1 parameter grammar and builds both role parts through
+ * the public `ShapeCreationService`, so the plugin never imports a concrete
+ * `Rect`/`Text` class. The parts belong to the same candidate-document edit as the
+ * container: a derived-id conflict or any part failure aborts the whole command and
+ * leaves no residue. `matches` claims only the explicit `g` + `data-kind="box"`
+ * marker; without this provider the codec falls back to the ordinary `g` provider
+ * and keeps the marker as imported metadata. `restore` rebuilds a complete subtree
+ * (children restored with their own world transforms and never re-transformed by
+ * the container) and rejects incomplete/wrongly typed boxes so a single read fails
+ * atomically instead of producing a partial domain object.
+ */
+export class BoxProvider implements ShapeProvider {
+  readonly type = 'box';
+  readonly usage = 'box <id> <x> <y> <w> <h> "label"';
+
+  matches(element: SvgElementView): boolean {
+    return element.tag === 'g' && element.attributes['data-kind'] === 'box';
+  }
+
+  create(request: ShapeCreationRequest, shapes: ShapeCreationService): Shape {
+    if (request.parameters.length !== 5)
+      throw new DomainError(`invalid parameters for ${this.type}`);
+    const x = this.number(request, 0);
+    const y = this.number(request, 1);
+    const width = this.number(request, 2);
+    const height = this.number(request, 3);
+    const label = request.parameters[4]!;
+    const box = new BoxShape(request.id);
+    box.insert(
+      shapes.create({
+        type: 'rect',
+        id: box.rectId(),
+        parameters: [
+          Numbers.format(x),
+          Numbers.format(y),
+          Numbers.format(width),
+          Numbers.format(height),
+          DEFAULT_FILL,
+        ],
+      }),
+    );
+    box.insert(
+      shapes.create({
+        type: 'text',
+        id: box.labelId(),
+        parameters: [
+          Numbers.format(x + LABEL_OFFSET_X),
+          Numbers.format(y + height / 2 + LABEL_BASELINE_OFFSET),
+          label,
+        ],
+      }),
+    );
+    return box;
+  }
+
+  restore(element: SvgElementView, shapes: ShapeRestoreService): Shape {
+    if (!element.id) throw new DomainError('invalid box structure: missing id');
+    const box = new BoxShape(element.id);
+    this.applyAttributes(box, element);
+    for (const child of element.children) box.insert(shapes.restore(child));
+    box.validateInvariant();
+    return box;
+  }
+
+  private number(request: ShapeCreationRequest, index: number): number {
+    return Numbers.parse(request.parameters[index]!, `parameter ${index + 1}`);
+  }
+
+  /**
+   * Applies already-normalized element attributes in Phase 1 order: presentation
+   * goes through the validated setters, everything else (including the domain
+   * marker and custom metadata) is preserved. The container never re-applies
+   * `element.transform`; children carry their own accumulated world transform.
+   */
+  private applyAttributes(box: BoxShape, element: SvgElementView): void {
+    for (const [name, value] of Object.entries(element.attributes)) {
+      if (name === 'id' || name.startsWith('xmlns')) continue;
+      if (name === 'fill' || name === 'stroke' || name === 'stroke-width')
+        box.setAttribute(name, value);
+      else box.preserveAttribute(name, value);
+    }
+  }
+}

+import { DomainError, Group } from '@svgframe/core';
+import type { Shape } from '@svgframe/core';
+
+/**
+ * Composite `box` type: SVG tag `g`, domain kind `box`, serialized with
+ * `data-kind="box"`.
+ *
+ * A box is a container with two necessary role parts whose derived ids are
+ * `<id>-rect` and `<id>-label`; any extra children (including nested boxes) are
+ * ordinary members. Role parts are resolved by id on every access, so `clone()`
+ * (inherited from the public `Group` clone path plus the `newContainer()` hook)
+ * automatically rebinds the copied box to its copied parts without sharing mutable
+ * children with the source snapshot.
+ */
+export class BoxShape extends Group {
+  constructor(id: string) {
+    super(id);
+    this.preserveAttribute('data-kind', 'box');
+  }
+  override get kind(): string {
+    return 'box';
+  }
+  rectId(): string {
+    return `${this.id}-rect`;
+  }
+  labelId(): string {
+    return `${this.id}-label`;
+  }
+  /** The necessary rectangle; throws when the composition is incomplete. */
+  rect(): Shape {
+    return this.part(this.rectId(), 'rect');
+  }
+  /** The necessary label; throws when the composition is incomplete. */
+  label(): Shape {
+    return this.part(this.labelId(), 'text');
+  }
+  /**
+   * Structural policy: the two role parts cannot be deleted or moved out on their
+   * own; deleting the whole box stays allowed because that decision belongs to its
+   * parent container.
+   */
+  override canRemoveChild(child: Shape): boolean {
+    return child.id !== this.rectId() && child.id !== this.labelId();
+  }
+  /** A box owns its internal composition, so `ungroup` must refuse it. */
+  override canUngroup(): boolean {
+    return false;
+  }
+  /**
+   * Public feedback hook: a created box reports its container tag plus the role
+   * parts, without the kernel knowing the `box` domain type.
+   */
+  override creationParts(): readonly string[] {
+    return [
+      `${this.tag}#${this.id}`,
+      ...this.children().map((child) => `${child.tag}#${child.id}`),
+    ];
+  }
+  /**
+   * Every node invariant: the serialization marker stays and both role parts are
+   * present with the required element names.
+   */
+  override validateInvariant(): void {
+    if (this.attributes()['data-kind'] !== 'box')
+      throw new DomainError(`invalid box: ${this.id} must keep data-kind=box`);
+    this.rect();
+    this.label();
+  }
+  /** Retains the actual plugin type while `Group.clone` copies members. */
+  protected override newContainer(): BoxShape {
+    return new BoxShape(this.id);
+  }
+  private part(id: string, tag: 'rect' | 'text'): Shape {
+    const child = this.children().find((candidate) => candidate.id === id);
+    if (!child) throw new DomainError(`incomplete box: ${this.id} is missing ${id}`);
+    if (child.tag !== tag) throw new DomainError(`invalid box: ${id} must be a <${tag}> part`);
+    return child;
+  }
+}

+import { Numbers, Shape } from '@svgframe/core';
+
+const X_KEYS = new Set(['x', 'cx', 'x1', 'x2']);
+const Y_KEYS = new Set(['y', 'cy', 'y1', 'y2']);
+
+/**
+ * Core-only leaf double for box plugin tests: it mimics the observable geometry,
+ * text and presentation behavior of the builtins leaves without importing them.
+ */
+export class TestLeafShape extends Shape {
+  readonly tag: string;
+  private readonly kindName: string;
+  private readonly values = new Map<string, number>();
+  private content: string | null;
+  constructor(
+    id: string,
+    tag: string,
+    values: Readonly<Record<string, number>> = {},
+    content: string | null = null,
+    kind: string = tag,
+  ) {
+    super(id);
+    this.tag = tag;
+    this.kindName = kind;
+    this.content = content;
+    for (const [name, value] of Object.entries(values)) this.values.set(name, value);
+  }
+  override get kind(): string {
+    return this.kindName;
+  }
+  override move(dx: number, dy: number): void {
+    Numbers.finite(dx, 'dx');
+    Numbers.finite(dy, 'dy');
+    for (const name of this.values.keys()) {
+      if (X_KEYS.has(name)) this.values.set(name, this.values.get(name)! + dx);
+      if (Y_KEYS.has(name)) this.values.set(name, this.values.get(name)! + dy);
+    }
+  }
+  override scale(factor: number): void {
+    Numbers.positive(factor, 'scale factor');
+    for (const [name, value] of this.values)
+      this.values.set(name, Numbers.finite(value * factor, name));
+  }
+  override clone(): TestLeafShape {
+    return new TestLeafShape(
+      this.id,
+      this.tag,
+      Object.fromEntries(this.values),
+      this.content,
+      this.kindName,
+    );
+  }
+  override geometry(): Readonly<Record<string, string>> {
+    return Object.freeze(
+      Object.fromEntries([...this.values].map(([name, value]) => [name, Numbers.format(value)])),
+    );
+  }
+  override textContent(): string | null {
+    return this.content;
+  }
+  override summary(): string {
+    return '';
+  }
+  protected override setSpecificAttribute(name: string, value: string): void {
+    if (name === 'text') {
+      this.content = value;
+      return;
+    }
+    super.setSpecificAttribute(name, value);
+  }
+}

+function leafFromParameters(request: ShapeCreationRequest): TestLeafShape {
